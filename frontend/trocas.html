async function enviarProposta() { const peer = document.getElementById('target-peer').value.trim(); const offer = document.getElementById('offer-sticker').value.trim(); const want = document.getElementById('want-sticker').value.trim(); if(!peer || !offer || !want) return alert("Preencha todos os campos!"); try { await fetch('/api/trade/offer', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ target_peer: peer, offered_sticker: offer, wanted_sticker: want }) }); alert(`Proposta enviada para ${peer}!`); } catch (err) { console.error(err); } } async function responderProposta(peerId, acao) { try { await fetch(`/api/trade/${acao}`, { // acao será 'accept' ou 'reject' method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ peer_id: peerId }) }); alert(`Proposta de ${peerId} foi ${acao === 'accept' ? 'aceita' : 'recusada'}!`); } catch (err) { console.error(err); } } // Checa propostas pendentes e desenha os botões na tela setInterval(async () => { try { const res = await fetch('/api/trade/pending'); const propostas = await res.json(); const lista = document.getElementById('propostas-recebidas'); if(propostas.length > 0) { lista.innerHTML = propostas.map(p => `
  • ${p.sender_peer_id} oferece ${p.offered_sticker} por sua ${p.wanted_sticker}.
    • `).join(''); } else { lista.innerHTML = '
  • Nenhuma proposta pendente no momento.
    • '; } } catch (e) {} }, 2000);
